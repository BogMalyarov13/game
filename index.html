import pygame import math import random import sys # Инициализация pygame.init() WIDTH, HEIGHT = 800, 600 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("Глаз ужаса: Лунтик") clock = pygame.time.Clock() font = pygame.font.Font(None, 36) # Цвета WHITE = (255, 255, 255) BLACK = (0, 0, 0) RED = (200, 0, 0) GREEN = (0, 200, 0) BLUE = (0, 100, 200) YELLOW = (255, 255, 0) PURPLE = (150, 0, 255) BROWN = (139, 69, 19) LUNTIK_COLOR = (255, 180, 200) # Игрок class Player: def __init__(self, x, y): self.x = x self.y = y self.angle = 0 # направление взгляда в градусах (0 = вправо) self.speed = 3 self.radius = 15 self.fear = 0 self.eyes_closed = False self.eye_close_timer = 0 def move(self, dx, dy): # Движение с учётом границ new_x = self.x + dx new_y = self.y + dy if 20 < new_x < WIDTH - 20: self.x = new_x if 20 < new_y < HEIGHT - 20: self.y = new_y def rotate(self, delta): self.angle = (self.angle + delta) % 360 def close_eyes(self): self.eyes_closed = True self.eye_close_timer = 60 # на 1 секунду (60 кадров при 60 fps) def update(self): if self.eyes_closed: self.eye_close_timer -= 1 if self.eye_close_timer <= 0: self.eyes_closed = False self.eye_close_timer = 0 # Страх медленно убывает, если не смотрим на монстра if not self.is_looking_at_monster(): self.fear = max(0, self.fear - 0.1) def is_looking_at_monster(self): # Проверяем, смотрит ли игрок на какое-либо зеркало с Лунтиком # Упрощённо: проверяем расстояние и угол до активного зеркала for mirror in mirrors: if mirror.active and mirror.luntik_visible: # Вектор от игрока к зеркалу dx = mirror.x - self.x dy = mirror.y - self.y dist = math.hypot(dx, dy) if dist > 200: continue # Угол до зеркала angle_to_mirror = math.degrees(math.atan2(-dy, dx)) # в pygame y вниз angle_to_mirror = angle_to_mirror % 360 # Разница между направлением взгляда и углом на зеркало diff = abs(angle_to_mirror - self.angle) if diff > 180: diff = 360 - diff if diff < 30: # если смотрит примерно на зеркало return True return False # Зеркало class Mirror: def __init__(self, x, y, width=40, height=60): self.x = x self.y = y self.width = width self.height = height self.active = True self.luntik_visible = False self.timer = 0 def update(self, player): # Лунтик появляется, если игрок смотрит на зеркало достаточно долго if self.active: if player.is_looking_at_monster() and not player.eyes_closed: self.timer += 1 if self.timer > 30: # полсекунды взгляда self.luntik_visible = True # увеличиваем страх player.fear = min(100, player.fear + 0.4) else: self.timer = max(0, self.timer - 1) if self.timer < 10: self.luntik_visible = False def draw(self, screen): if not self.active: return # Рисуем зеркало (серый прямоугольник) pygame.draw.rect(screen, (100, 100, 100), (self.x - self.width//2, self.y - self.height//2, self.width, self.height)) pygame.draw.rect(screen, WHITE, (self.x - self.width//2 + 2, self.y - self.height//2 + 2, self.width-4, self.height-4), 2) if self.luntik_visible: # Рисуем Лунтика в зеркале # Голова pygame.draw.circle(screen, LUNTIK_COLOR, (self.x, self.y), 20) # Глаза - большие, жуткие pygame.draw.circle(screen, WHITE, (self.x - 8, self.y - 5), 8) pygame.draw.circle(screen, WHITE, (self.x + 8, self.y - 5), 8) pygame.draw.circle(screen, BLACK, (self.x - 10, self.y - 3), 4) pygame.draw.circle(screen, BLACK, (self.x + 6, self.y - 3), 4) # Улыбка pygame.draw.arc(screen, BLACK, (self.x-12, self.y-5, 24, 20), 0.1, math.pi-0.1, 3) # Создаём объекты player = Player(WIDTH//2, HEIGHT//2) mirrors = [ Mirror(150, 150), Mirror(650, 150), Mirror(150, 450), Mirror(650, 450), ] # Генерация случайных стен (препятствия) walls = [] for _ in range(8): x = random.randint(100, 700) y = random.randint(100, 500) w = random.randint(30, 100) h = random.randint(30, 80) walls.append(pygame.Rect(x, y, w, h)) # Главный игровой цикл running = True while running: screen.fill(BLACK) # События for event in pygame.event.get(): if event.type == pygame.QUIT: running = False if event.type == pygame.KEYDOWN: if event.key == pygame.K_q: player.close_eyes() if event.key == pygame.K_ESCAPE: running = False # Управление keys = pygame.key.get_pressed() dx, dy = 0, 0 if keys[pygame.K_a]: dx = -player.speed if keys[pygame.K_d]: dx = player.speed if keys[pygame.K_w]: dy = -player.speed if keys[pygame.K_s]: dy = player.speed # Поворот (влево/вправо) if keys[pygame.K_LEFT]: player.rotate(-3) if keys[pygame.K_RIGHT]: player.rotate(3) player.move(dx, dy) # Обновление for mirror in mirrors: mirror.update(player) player.update() # Проверка проигрыша if player.fear >= 100: screen.fill(RED) text = font.render("Вас поглотил Лунтик! Нажмите ESC", True, WHITE) screen.blit(text, (WIDTH//2 - 200, HEIGHT//2)) pygame.display.flip() # Ждём ESC waiting = True while waiting: for e in pygame.event.get(): if e.type == pygame.QUIT: pygame.quit() sys.exit() if e.type == pygame.KEYDOWN and e.key == pygame.K_ESCAPE: waiting = False running = False continue # Отрисовка # Стены for wall in walls: pygame.draw.rect(screen, BROWN, wall) # Зеркала for mirror in mirrors: mirror.draw(screen) # Игрок (круг) pygame.draw.circle(screen, BLUE, (int(player.x), int(player.y)), player.radius) # Линия взгляда end_x = player.x + 40 * math.cos(math.radians(player.angle)) end_y = player.y - 40 * math.sin(math.radians(player.angle)) pygame.draw.line(screen, YELLOW, (player.x, player.y), (end_x, end_y), 3) # Если глаза закрыты - затемнение if player.eyes_closed: s = pygame.Surface((WIDTH, HEIGHT), pygame.SRCALPHA) s.fill((0, 0, 0, 180)) screen.blit(s, (0, 0)) text = font.render("Глаза закрыты", True, WHITE) screen.blit(text, (WIDTH//2 - 80, 30)) # Шкала страха pygame.draw.rect(screen, (50, 50, 50), (20, 20, 200, 30)) pygame.draw.rect(screen, RED, (20, 20, 2 * player.fear, 30)) text = font.render(f"Страх: {int(player.fear)}%", True, WHITE) screen.blit(text, (230, 20)) # Информация info = font.render("WASD - движение, стрелки - взгляд, Q - закрыть глаза", True, WHITE) screen.blit(info, (20, HEIGHT - 40)) pygame.display.flip() clock.tick(60) pygame.quit() sys.exit()
